cmake_minimum_required(VERSION 3.18)
project(MasterAgentInventoryExample LANGUAGES CXX)
if(NOT TARGET MasterAgent::Core)
    find_package(MasterAgent CONFIG REQUIRED)
endif()
find_package(CURL REQUIRED)
add_executable(inventory_agent main.cpp)
target_compile_features(inventory_agent PRIVATE cxx_std_17)
target_link_libraries(inventory_agent PRIVATE MasterAgent::Core CURL::libcurl)
